package fuzz

import (
 "context"
 "encoding/json"
 "fmt"
 "os"
 "reflect"
 "sync"
 "time"
)

// Isolated diagnostics only. Equality, not Error()/errors.Is, determines IDs.
var diagnosticErrors = struct {sync.Mutex; ids map[error]string}{ids:make(map[error]string)}
func DiagnosticErrorIdentity(err error) string {
 if err==nil {return "nil"}
 if err==context.DeadlineExceeded {return "deadline"}
 if err==context.Canceled {return "canceled"}
 if !reflect.ValueOf(err).Comparable() {return fmt.Sprintf("unsupported:%T",err)}
 diagnosticErrors.Lock();defer diagnosticErrors.Unlock()
 if id,ok:=diagnosticErrors.ids[err];ok {return id}
 id:=fmt.Sprintf("error:%T#%d",err,len(diagnosticErrors.ids)+1)
 diagnosticErrors.ids[err]=id
 return id
}
func DiagnosticParentLabel(err error,read bool) string {
 if !read {return "unread"}
 return DiagnosticErrorIdentity(err)
}
func DiagnosticObservationLabels(event string,child,parent,old error,parentRead bool)(string,string,string,bool) {
 if event=="return" {return "unread",DiagnosticParentLabel(parent,parentRead),"unread",false}
 return DiagnosticErrorIdentity(child),DiagnosticParentLabel(parent,parentRead),DiagnosticErrorIdentity(old),true
}
type diagnosticTrace struct {
 Schema int `json:"schema"`
 Seq int `json:"seq"`
 Mode string `json:"mode"`
 Event string `json:"event"`
 Source string `json:"source"`
 Incoming string `json:"incoming"`
 ChildErr string `json:"childErr"`
 ParentRead bool `json:"parentRead"`
 ParentErr string `json:"parentErr"`
 OldError string `json:"oldError"`
 NewError string `json:"newError"`
 Stopping bool `json:"stopping"`
 StoppingRead bool `json:"stoppingRead"`
 GateEntered bool `json:"gateEntered"`
 Propagated bool `json:"propagated"`
 Watchdog bool `json:"watchdog"`
 ActiveWorkers int `json:"activeWorkers"`
}
type diagnosticFuzzState struct {
 mode string
 gate *context.DiagnosticFuzzGate
 seq int
}
func diagnosticStart(ctx context.Context) *diagnosticFuzzState {
 mode:=os.Getenv("TALENRO_FUZZ_DIAG_MODE")
 if mode=="" {return nil}
 switch mode {case "observe","parent-first","synchronized","cancel": default:panic("unsupported diagnostic mode")}
 d:=&diagnosticFuzzState{mode:mode,gate:context.DiagnosticFuzzRegister(ctx)}
 if mode=="observe" {d.gate.Release()}
 if mode=="cancel" {time.AfterFunc(500*time.Millisecond,d.gate.Cancel)}
 return d
}
func(d *diagnosticFuzzState) beforeDeadline() {
 if d==nil || d.mode=="observe" {return}
 <-d.gate.Entered()
 if d.mode=="synchronized" {d.gate.Release();<-d.gate.Propagated()}
}
func diagnosticClosed(ch <-chan struct{}) bool {select {case <-ch:return true;default:return false}}
func(d *diagnosticFuzzState) release() {if d!=nil {d.gate.Release()}}
func(d *diagnosticFuzzState) emit(event,source string,incoming,child,parent,old,next error,parentRead,stopping bool,workers int) {
 if d==nil {return}
 d.seq++
 childLabel,parentLabel,oldLabel,stoppingRead:=DiagnosticObservationLabels(event,child,parent,old,parentRead)
 r:=diagnosticTrace{Schema:1,Seq:d.seq,Mode:d.mode,Event:event,Source:source,
 Incoming:DiagnosticErrorIdentity(incoming),ChildErr:childLabel,
 ParentRead:parentRead,ParentErr:parentLabel,OldError:oldLabel,NewError:DiagnosticErrorIdentity(next),
 Stopping:stopping,StoppingRead:stoppingRead,GateEntered:diagnosticClosed(d.gate.Entered()),Propagated:diagnosticClosed(d.gate.Propagated()),Watchdog:d.gate.WatchdogExpired(),ActiveWorkers:workers}
 data,err:=json.Marshal(r)
 if err!=nil {panic(err)}
 if _,err=fmt.Fprintf(os.Stderr,"TALENRO_FUZZ_DIAG %s\n",data);err!=nil {panic(err)}
}
